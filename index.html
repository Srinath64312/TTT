# Tail-recursive Minimax evaluator without ANY loops
def _evaluate_moves_recursive(b, depth, is_max, moves, best_val, best_move):
    if not moves:
        return best_val, best_move
    pos = moves[0]
    next_b = make_move(b, pos, "O" if is_max else "X")
    score, _ = minimax(next_b, depth + 1, not is_max)
    new_best_val = max(best_val, score) if is_max else min(best_val, score)
    new_best_move = pos if new_best_val == score else best_move
    return _evaluate_moves_recursive(b, depth, is_max, moves[1:], new_best_val, new_best_move)
